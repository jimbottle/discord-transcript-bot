# Apple Silicon GPU (Metal) transcription backend (discord-transcript-bot-d8g).
# Selected automatically on Darwin/arm64 (ASR_BACKEND=auto); elsewhere the bot
# falls back to faster-whisper, so these are scoped to Apple Silicon to keep
# Linux/CI installs unaffected. scipy gives an anti-aliased 48k->16k resample
# (protects WER) for the MLX path.
mlx-whisper; sys_platform == 'darwin' and platform_machine == 'arm64'
scipy; sys_platform == 'darwin' and platform_machine == 'arm64'

# output_config.effort, which older SDKs reject with a TypeError at call time.
